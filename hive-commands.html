Hive Commands

1. Start Hive Shell

hive

2. Show Databases

show databases;

3. Create Database

create database mydb;

4. Use Database

use mydb;

5. Show Tables

show tables;

6. Create Table


create table emp(id int, name string)
row format delimited fields terminated by ',';

7. Load Data

load data local inpath 'file.csv' into table emp;

8. Select Query

select * from emp;

9. Describe Table

describe emp;

10. Drop Table

drop table emp;

11. Create External Table


create external table logs(line string)
location '/data/logs/';

12. Partition Table


create table sales(id int, amount int)
partitioned by (year int)
row format delimited fields terminated by ',';

13. Add Partition

alter table sales add partition(year=2024);

14. Hive Beeline

beeline -u jdbc:hive2://localhost:10000
