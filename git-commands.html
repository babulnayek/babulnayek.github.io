Git Commands (Complete Reference)

1. Basic Git Commands

git init
git clone https://github.com/user/repo.git
git status
git add file.txt
git add .
git commit -m "message"
git push origin main
git pull

2. Branching Commands

git branch
git branch feature/login
git checkout feature/login
git checkout -b feature/api
git merge feature/login
git branch -d feature/login

3. View Commit History

git log
git log --oneline
git log --graph --decorate --all
git show COMMIT_ID

4. Diff Commands

git diff
git diff --staged
git diff main..feature/login

5. Stash Commands

git stash
git stash list
git stash apply
git stash pop
git stash drop

6. Reset Commands

git reset file.txt
git reset --soft HEAD~1
git reset --mixed HEAD~1
git reset --hard HEAD~1

7. Revert Commands

git revert COMMIT_ID
git revert -n COMMIT_ID

8. Cherry Pick Commands

git cherry-pick COMMIT_ID
git cherry-pick A..B

9. Rebase Commands

git rebase main
git rebase -i HEAD~5
git rebase --abort
git rebase --continue

10. Tagging Commands

git tag v1.0
git tag -a v1.0 -m "Release 1.0"
git push origin v1.0

11. Remote Commands

git remote -v
git remote add origin https://github.com/user/repo.git
git remote remove origin

12. Clean Working Directory

git clean -n
git clean -f
git clean -fd

13. GitHub SSH Commands

ssh-keygen -t rsa -b 4096
ssh-copy-id user@host
cat ~/.ssh/id_rsa.pub

14. Submodule Commands

git submodule add https://github.com/user/repo.git
git submodule update --init --recursive

15. Troubleshooting Commands

git fsck
git gc
git clean -fd
git reset --hard

16. Enterprise DevOps Git Commands

git fetch --all
git pull --rebase
git push --force-with-lease
git shortlog -sn
git blame file.txt
