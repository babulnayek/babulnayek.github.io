GitHub Actions Commands

1. Create Workflow

.github/workflows/main.yml

2. Trigger Workflow Manually

workflow_dispatch:

3. Trigger on Push


on:
  push:
    branches:
      - main

4. Trigger on Pull Request


on:
  pull_request:
    branches:
      - main

5. Run Shell Commands


steps:
  - name: Run script
    run: echo "Hello"

6. Checkout Code


- uses: actions/checkout@v3

7. Setup Node


- uses: actions/setup-node@v3
  with:
    node-version: 18

8. Upload Artifact


- uses: actions/upload-artifact@v3
  with:
    name: build
    path: dist/

9. Download Artifact


- uses: actions/download-artifact@v3
  with:
    name: build

10. Matrix Build


strategy:
  matrix:
    os: [ubuntu-latest, windows-latest]
