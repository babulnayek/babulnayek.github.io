Unix Commands (Complete Reference)

1. Basic File Listing

ls -l
ls -la

2. Change Directory

cd /path/to/dir

3. View File

cat file.txt
tail -n 50 file.txt
head -n 50 file.txt

4. Search in File

grep "error" file.txt
grep -r "pattern" /path

5. Disk Usage

df -h
du -sh /path

6. Permissions

chmod 755 script.sh
chown user:group file.txt

7. Processes

ps -ef
top
kill -9 PID

8. Advanced File & Directory Operations

find /var/log -type f -mtime -1
find / -type f -size +500M
find /data -name "*.log" -exec rm -f {} \;
rsync -avz /src/ /dest/
rsync -avz --delete /src/ /dest/
tar -czvf backup.tar.gz /data
tar -xzvf backup.tar.gz
du -sh * | sort -h
df -hT

9. Advanced Process Management

ps -eo pid,ppid,cmd,%mem,%cpu --sort=-%cpu
top -o %MEM
htop
kill -9 $(pidof java)
pkill -f spark
nohup python app.py &
jobs -l
fg %1

10. Advanced Networking

netstat -tulnp
ss -tulnp
curl -I https://google.com
wget -c https://file.com/bigfile.iso
ping -c 5 google.com
traceroute google.com
nslookup google.com
dig google.com
telnet host 8080

11. Disk & Storage Management

lsblk
fdisk -l
mount /dev/sdb1 /mnt/data
umount /mnt/data
lsof /mnt/data
iostat -xz 1
vmstat 1

12. User & Permission Management

chmod 4755 script.sh
chown -R user:group /data
useradd -m bharat
passwd bharat
visudo
sudo -l

13. Log Analysis

tail -f /var/log/messages
grep -R "ERROR" /var/log/
grep -R "timeout" /var/log/app.log | wc -l
awk '{print $1,$5}' /var/log/app.log
sed -n '1,50p' file.log
cut -d',' -f2 file.csv

14. System Monitoring

uptime
dmesg | tail
free -m
sar -u 1 5
sar -n DEV 1 5
mpstat -P ALL 1

15. Cron & Automation

crontab -l
crontab -e
0 2 * * * /usr/bin/python3 /scripts/backup.py
/etc/init.d/cron restart

16. Package Management

yum install httpd
yum update
apt-get install nginx
apt-get update && apt-get upgrade

17. Advanced Shell Scripting

for i in *.log; do echo $i; done
if [ -f file.txt ]; then echo "exists"; fi
while read line; do echo $line; done < file.txt
export PATH=$PATH:/opt/bin

18. Security Commands

ssh-keygen -t rsa
ssh-copy-id user@host
chmod 600 ~/.ssh/id_rsa
fail2ban-client status

19. Advanced Tools

tmux
screen
nc -lvp 4444
strace -p PID
lsof -i :8080
fuser -k 8080/tcp
